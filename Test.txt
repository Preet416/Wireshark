1) Practical 1 : Monoalphabetic Cipher
Aim : Write a program to implement a Monoalphabetic Cipher in Python for
encryption and decryption using a substitution key.

import string

letters = string.ascii_lowercase

key = "qwertyuiopasdfghjklzxcvbnm"

encrypt_dict = dict(zip(letters, key))
decrypt_dict = dict(zip(key, letters))

def encrypt(plaintext):
    plaintext = plaintext.lower()
    ciphertext = ""

    for char in plaintext:
        if char in encrypt_dict:
            ciphertext += encrypt_dict[char]
        else:
            ciphertext += char

    return ciphertext

def decrypt(ciphertext):
    ciphertext = ciphertext.lower()
    plaintext = ""

    for char in ciphertext:
        if char in decrypt_dict:
            plaintext += decrypt_dict[char]
        else:
            plaintext += char

    return plaintext

message = "hello world"

encrypted = encrypt(message)
decrypted = decrypt(encrypted)

print("Original :", message)
print("Encrypted:", encrypted)
print("Decrypted:", decrypted)

2) Practical 2 : Playfair Cipher
B] Aim: To implement a Playfair Cipher in Python for encrypting and decrypting
text using a key-based digraph substitution method.
Code :

def generate_key_matrix(key):
    key = key.upper().replace("J", "I")
    matrix = []
    used = set()

    for char in key:
        if char not in used and char.isalpha():
            used.add(char)
            matrix.append(char)

    for char in "ABCDEFGHIKLMNOPQRSTUVWXYZ":
        if char not in used:
            matrix.append(char)

    return [matrix[i:i+5] for i in range(0, 25, 5)]

def find_position(matrix, char):
    for i in range(5):
        for j in range(5):
            if matrix[i][j] == char:
                return i, j

def prepare_text(text):
    text = text.upper().replace("J", "I").replace(" ", "")
    result = ""
    i = 0

    while i < len(text):
        a = text[i]
        b = text[i+1] if i+1 < len(text) else "X"

        if a == b:
            result += a + "X"
            i += 1
        else:
            result += a + b
            i += 2

    if len(result) % 2 != 0:
        result += "X"

    return result

def encrypt(text, matrix):
    text = prepare_text(text)
    cipher = ""

    for i in range(0, len(text), 2):
        a, b = text[i], text[i+1]

        r1, c1 = find_position(matrix, a)
        r2, c2 = find_position(matrix, b)

        if r1 == r2:
            cipher += matrix[r1][(c1+1)%5]
            cipher += matrix[r2][(c2+1)%5]

        elif c1 == c2:
            cipher += matrix[(r1+1)%5][c1]
            cipher += matrix[(r2+1)%5][c2]

        else:
            cipher += matrix[r1][c2]
            cipher += matrix[r2][c1]

    return cipher

def decrypt(text, matrix):
    plain = ""

    for i in range(0, len(text), 2):
        a, b = text[i], text[i+1]

        r1, c1 = find_position(matrix, a)
        r2, c2 = find_position(matrix, b)

        if r1 == r2:
            plain += matrix[r1][(c1-1)%5]
            plain += matrix[r2][(c2-1)%5]

        elif c1 == c2:
            plain += matrix[(r1-1)%5][c1]
            plain += matrix[(r2-1)%5][c2]

        else:
            plain += matrix[r1][c2]
            plain += matrix[r2][c1]

    return plain

key = "keyword"

matrix = generate_key_matrix(key)

message = "hello world"

encrypted = encrypt(message, matrix)
decrypted = decrypt(encrypted, matrix)

print("Key Matrix:")
for row in matrix:
    print(row)

print("\nOriginal :", message)
print("Encrypted:", encrypted)
print("Decrypted:", decrypted)

3) Practical 3 : Hill Cipher
C] Aim: To implement a Hill Cipher in Python for encrypting and decrypting text
using matrix-based substitution
Code :

import numpy as np

def text_to_numbers(text):
    return [ord(char) - ord('A') for char in text]

def numbers_to_text(numbers):
    return ''.join(chr(num % 26 + ord('A')) for num in numbers)

def encrypt(plaintext, key_matrix):
    plaintext = plaintext.upper().replace(" ", "")

    if len(plaintext) % 2 != 0:
        plaintext += 'X'

    cipher_text = ""

    for i in range(0, len(plaintext), 2):
        pair = text_to_numbers(plaintext[i:i+2])
        pair_vector = np.array(pair).reshape(2, 1)

        result = np.dot(key_matrix, pair_vector) % 26
        cipher_text += numbers_to_text(result.flatten())

    return cipher_text

def mod_inverse(a, m):
    for i in range(1, m):
        if (a * i) % m == 1:
            return i
    return None

def decrypt(ciphertext, key_matrix):

    det = int(round(np.linalg.det(key_matrix)))
    det = det % 26

    det_inv = mod_inverse(det, 26)

    adj = np.array([
        [key_matrix[1][1], -key_matrix[0][1]],
        [-key_matrix[1][0], key_matrix[0][0]]
    ])

    inv_key = (det_inv * adj) % 26

    plain_text = ""

    for i in range(0, len(ciphertext), 2):
        pair = text_to_numbers(ciphertext[i:i+2])
        pair_vector = np.array(pair).reshape(2, 1)

        result = np.dot(inv_key, pair_vector) % 26
        plain_text += numbers_to_text(result.flatten())

    return plain_text

key_matrix = np.array([
    [3, 3],
    [2, 5]
])

message = "HELLO"

encrypted = encrypt(message, key_matrix)
decrypted = decrypt(encrypted, key_matrix)

print("Original :", message)
print("Encrypted:", encrypted)
print("Decrypted:", decrypted)

def generate_key(text, key):
    key = list(key)

    if len(text) == len(key):
        return "".join(key)

    else:
        for i in range(len(text) - len(key)):
            key.append(key[i % len(key)])

    return "".join(key)

def encrypt(plaintext, key):
    plaintext = plaintext.upper().replace(" ", "")
    key = generate_key(plaintext, key.upper())

    cipher_text = ""

    for i in range(len(plaintext)):
        p = ord(plaintext[i]) - ord('A')
        k = ord(key[i]) - ord('A')

        c = (p + k) % 26
        cipher_text += chr(c + ord('A'))

    return cipher_text

def decrypt(ciphertext, key):
    key = generate_key(ciphertext, key.upper())

    plain_text = ""

    for i in range(len(ciphertext)):
        c = ord(ciphertext[i]) - ord('A')
        k = ord(key[i]) - ord('A')

        p = (c - k + 26) % 26
        plain_text += chr(p + ord('A'))

    return plain_text

5)  Practical 5 : Vernam Cipher
E] Aim : To implement a Vernam Cipher in Python for encrypting and decrypting
text using a one-time key and XOR operation.

def text_to_binary(text):
    return ''.join(format(ord(c), '08b') for c in text)

def binary_to_text(binary):
    chars = [binary[i:i+8] for i in range(0, len(binary), 8)]
    return ''.join(chr(int(b, 2)) for b in chars)

def xor(a, b):
    return ''.join('0' if i == j else '1' for i, j in zip(a, b))

def encrypt(plaintext, key):
    binary_text = text_to_binary(plaintext)
    binary_key = text_to_binary(key)

    cipher_binary = xor(binary_text, binary_key)

    return cipher_binary

def decrypt(cipher_binary, key):
    binary_key = text_to_binary(key)

    plain_binary = xor(cipher_binary, binary_key)

    return binary_to_text(plain_binary)

message = "HELLO"
key = "XMCKL"

encrypted = encrypt(message, key)
decrypted = decrypt(encrypted, key)

print("Original :", message)
print("Key :", key)
print("Encrypted (binary):", encrypted)
print("Decrypted:", decrypted)


6) Practical 6 : RSA Algorithm

import math

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def mod_inverse(e, phi):
    for d in range(1, phi):
        if (e * d) % phi == 1:
            return d

def generate_keys():
    p = 17
    q = 11

    n = p * q
    phi = (p - 1) * (q - 1)

    e = 2

    while gcd(e, phi) != 1:
        e += 1

    d = mod_inverse(e, phi)

    return (e, n), (d, n)

def encrypt(public_key, plaintext):
    e, n = public_key
    return [pow(ord(char), e, n) for char in plaintext]

def decrypt(private_key, ciphertext):
    d, n = private_key
    return ''.join(chr(pow(char, d, n)) for char in ciphertext)

public_key, private_key = generate_keys()

message = "HELLO"

encrypted = encrypt(public_key, message)
decrypted = decrypt(private_key, encrypted)

print("Public Key :", public_key)
print("Private Key:", private_key)
print("Original :", message)

def encrypt(plaintext, shift):
    result = ""

    for char in plaintext:
        if char.isalpha():

            base = ord('A') if char.isupper() else ord('a')

            result += chr((ord(char) - base + shift) % 26 + base)

        else:
            result += char

    return result

def decrypt(ciphertext, shift):
    return encrypt(ciphertext, -shift)

8) Practical 8 : Feistel Cipher
Aim : To implement a Feistel Cipher in Python for encrypting and decrypting data
using symmetric key rounds.
Code :

def xor(a, b):
    return format(int(a, 2) ^ int(b, 2), '04b')

def f(right, key):
    return xor(right, key)

def encrypt(plaintext, keys):

    pt = format(ord(plaintext), '08b')

    left = pt[:4]
    right = pt[4:]

    for key in keys:
        new_left = right
        new_right = xor(left, f(right, key))

        left, right = new_left, new_right

    return left + right

def decrypt(ciphertext, keys):

    left = ciphertext[:4]
    right = ciphertext[4:]

    for key in reversed(keys):
        new_right = left
        new_left = xor(right, f(left, key))

        left, right = new_left, new_right

    binary = left + right

    return chr(int(binary, 2))

keys = ["1010", "0101"]

message = "A"

encrypted = encrypt(message, keys)
decrypted = decrypt(encrypted, keys)

print("Original :", message)
print("Encrypted (binary):", encrypted)
print("Decrypted:", decrypted)
